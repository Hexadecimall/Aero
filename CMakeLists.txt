cmake_minimum_required(VERSION 3.20)
project(Aero VERSION 0.1.0 LANGUAGES CXX)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS src/main.cpp src/instructions.hpp)
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp aeroRuntimeSource)
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/src/instructions.hpp aeroInstructionSource)
string(REPLACE "#pragma once" "" aeroInstructionSource "${aeroInstructionSource}")
string(REPLACE "#include \"instructions.hpp\"" "${aeroInstructionSource}" aeroRuntimeSource "${aeroRuntimeSource}")
configure_file(src/embedded_runtime.hpp.in generated/embedded_runtime.hpp @ONLY)
add_executable(aero src/main.cpp src/aot.cpp)
target_include_directories(aero PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_compile_features(aero PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(aero PRIVATE /W4)
else()
  target_compile_options(aero PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(GNUInstallDirs)
install(TARGETS aero RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES README.md CONTRIBUTING.md LICENSE DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(FILES docs/building.md DESTINATION ${CMAKE_INSTALL_DOCDIR}/docs)

option(BUILD_TESTING "Enable Aero tests" OFF)
include(CTest)
if(BUILD_TESTING)
  add_test(NAME cli-help COMMAND aero --help)
  add_test(NAME cli-version COMMAND aero --version)
endif()
if(BUILD_TESTING AND UNIX)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  find_program(aeroClang NAMES clang REQUIRED)
  find_program(aeroLld NAMES ld.lld REQUIRED)
  add_test(NAME linux-aarch64-program
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/program.py
      $<TARGET_FILE:aero> ${aeroClang} ${aeroLld}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/echo.S ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
  add_test(NAME compiled-linux-programs
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/compiledPrograms.py
      $<TARGET_FILE:aero> ${aeroClang} ${aeroLld}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests ${CMAKE_CURRENT_BINARY_DIR}/compiledFixtures)
  set_tests_properties(linux-aarch64-program compiled-linux-programs PROPERTIES TIMEOUT 180)
endif()

set(CPACK_PACKAGE_NAME Aero)
set(CPACK_PACKAGE_VENDOR Aero)
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Linux program execution and native translation")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_FILE_NAME "aero-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
set(CPACK_GENERATOR "TGZ;ZIP")
set(CPACK_SOURCE_GENERATOR TGZ)
set(CPACK_SOURCE_IGNORE_FILES "/build[^/]*/;/\\.git/;/\\.github/;\\.o$;\\.DS_Store$;CMakeUserPresets.json;__pycache__")
include(CPack)
