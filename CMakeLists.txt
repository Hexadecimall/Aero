cmake_minimum_required(VERSION 3.20)
project(Aero VERSION 0.1.0 LANGUAGES C CXX)
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/generated)
if(NOT UNIX)
  set(aeroRuntimeSource "")
  configure_file(src/embedded_runtime.hpp.in generated/embedded_runtime.hpp @ONLY)
endif()
add_executable(aero src/main.cpp src/aot.cpp)
target_include_directories(aero PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_compile_features(aero PRIVATE cxx_std_20)
if(UNIX)
  find_package(LLVM CONFIG REQUIRED)
  if(LLVM_VERSION_MAJOR LESS 18)
    message(FATAL_ERROR "Aero requires LLVM, Clang, and LLD version 18 or newer")
  endif()
  find_package(Clang CONFIG REQUIRED HINTS "${LLVM_TOOLS_BINARY_DIR}/../lib/cmake/clang")
  find_package(LLD CONFIG REQUIRED HINTS "${LLVM_TOOLS_BINARY_DIR}/../lib/cmake/lld")
  find_program(aeroBuildClang NAMES clang++ HINTS "${LLVM_TOOLS_BINARY_DIR}" NO_DEFAULT_PATH REQUIRED)
  set(aeroRuntimeFlags -std=c++20 -E -P -DaeroRuntimeOnly -x c++)
  if(APPLE)
    if(NOT CMAKE_OSX_SYSROOT)
      execute_process(COMMAND xcrun --show-sdk-path OUTPUT_VARIABLE aeroSdk OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    else()
      set(aeroSdk "${CMAKE_OSX_SYSROOT}")
    endif()
    list(APPEND aeroRuntimeFlags -isysroot "${aeroSdk}" -mmacosx-version-min=13.0)
  endif()
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/generated/embedded_runtime.hpp
    COMMAND ${aeroBuildClang} ${aeroRuntimeFlags} ${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp
      -o ${CMAKE_CURRENT_BINARY_DIR}/generated/aeroRuntime.ii
    COMMAND ${CMAKE_COMMAND} -Dinput=${CMAKE_CURRENT_BINARY_DIR}/generated/aeroRuntime.ii
      -Doutput=${CMAKE_CURRENT_BINARY_DIR}/generated/embedded_runtime.hpp
      -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/embedRuntime.cmake
    DEPENDS src/main.cpp src/instructions.hpp cmake/embedRuntime.cmake CMakeLists.txt VERBATIM)
  target_sources(aero PRIVATE src/nativeCompiler.cpp ${CMAKE_CURRENT_BINARY_DIR}/generated/embedded_runtime.hpp)
  target_include_directories(aero SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS} ${LLD_INCLUDE_DIRS})
  target_link_libraries(aero PRIVATE clang-cpp LLVM lldCommon)
  if(APPLE)
    target_link_libraries(aero PRIVATE lldMachO)
  else()
    target_link_libraries(aero PRIVATE lldELF)
  endif()
endif()
if(MSVC)
  target_compile_options(aero PRIVATE /W4)
else()
  target_compile_options(aero PRIVATE -Wall -Wextra -Wpedantic)
  set_source_files_properties(src/aot.cpp PROPERTIES COMPILE_OPTIONS -Wno-overlength-strings)
endif()
include(GNUInstallDirs)
install(TARGETS aero RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES README.md CONTRIBUTING.md LICENSE DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(FILES docs/building.md DESTINATION ${CMAKE_INSTALL_DOCDIR}/docs)
if(UNIX)
  set_target_properties(aero PROPERTIES INSTALL_RPATH "${LLVM_LIBRARY_DIRS};${LLD_INSTALL_PREFIX}/lib")
endif()

option(BUILD_TESTING "Enable Aero tests" OFF)
include(CTest)
if(BUILD_TESTING)
  add_test(NAME cli-help COMMAND aero --help)
  add_test(NAME cli-version COMMAND aero --version)
endif()
if(BUILD_TESTING AND UNIX)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  find_program(aeroClang NAMES clang REQUIRED)
  find_program(aeroLld NAMES ld.lld REQUIRED)
  add_test(NAME linux-aarch64-program
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/program.py
      $<TARGET_FILE:aero> ${aeroClang} ${aeroLld}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/echo.S ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
  add_test(NAME compiled-linux-programs
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/compiledPrograms.py
      $<TARGET_FILE:aero> ${aeroClang} ${aeroLld}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests ${CMAKE_CURRENT_BINARY_DIR}/compiledFixtures)
  set_tests_properties(linux-aarch64-program compiled-linux-programs PROPERTIES TIMEOUT 180)
  add_test(NAME export-without-tools
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/exportWithoutTools.py
      $<TARGET_FILE:aero> ${aeroClang} ${aeroLld}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/echo.S ${CMAKE_CURRENT_BINARY_DIR}/isolatedExport)
  set_tests_properties(export-without-tools PROPERTIES TIMEOUT 120)
endif()

set(CPACK_PACKAGE_VENDOR Aero)

set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_FILE_NAME "aero-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
set(CPACK_GENERATOR "TGZ;ZIP")
set(CPACK_SOURCE_GENERATOR TGZ)
set(CPACK_SOURCE_IGNORE_FILES "/build[^/]*/;/\\.git/;/\\.github/;\\.o$;\\.DS_Store$;CMakeUserPresets.json;__pycache__")
